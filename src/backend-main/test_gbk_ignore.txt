        BeanUtils.copyProperties(current, newVersion);  // 婢跺秴鍩楅崢鐔洪張鐫橀幀?

        }



        if (entity.getNamespaceUri() != null) {



            newVersion.setNamespaceUri(entity.getNamespaceUri());



        }



        // 閻㈢喐鍨氶弬澧楅張褰块敍鍫濐洤1.0 -> 1.1?

    @Override



    @Transactional



    public boolean importOntology(MultipartFile excelFile, String projectName, String creator) {



        try {



            // 1. 妤犲矁鐦夐弬鍥︾猾璇



            String fileName = excelFile.getOriginalFilename();



            if (fileName == null || (!fileName.endsWith(".xlsx") && !fileName.endsWith(".xls"))) {



                log.error("鐎电厧鍙嗘径瑙﹂敍灞肩矌顖涘瘮Excel閺傚洣");



                throw new RuntimeException("娴犲懏?xlsx?xls閺嶇厧绱￠惃Excel閺傚洣");



            }







            // 2. 閺屻儲鏋冩禒鑸垫Ц閸氾缚璐熺粚?



            if (excelFile.isEmpty()) {



                throw new RuntimeException("閺傚洣娆㈡稉宥堝厴娑撹櫣");



            }







            // 3. 鐟欙絾鐎紼xcel閺傚洣



            InputStream inputStream = excelFile.getInputStream();



            Workbook workbook = WorkbookFactory.create(inputStream);



            Sheet sheet = workbook.getSheetAt(0);  // 閸欐牜顑囨稉娑搒heet



            Row firstDataRow = sheet.getRow(1);  // ?鐞涘奔璐熺悰銊ャ仈閿涘瞼1鐞涘奔璐熼弫鐗堝祦







            // 4. 娴燛xcel娑擃叀顕伴崣銆嶉惄顔兼倳缁夊府绱欐导妯哄帥娴ｈ法Excel娑擃厾娈戦崥宥囆為敍?



            String excelProjectName = projectName;



            if (firstDataRow != null && firstDataRow.getCell(1) != null) {



                excelProjectName = firstDataRow.getCell(1).getStringCellValue();



            }



            if (StringUtils.isEmpty(excelProjectName)) {



                throw new RuntimeException("妞ゅ湱娲伴崥宥囆炴稉宥堝厴娑撹櫣");



            }







            // 5. 閺屻儵銆嶉惄顔芥Ц閸氾箑鍑℃ê



            boolean exists = checkNameExists(excelProjectName, null);



            if (exists) {



                // 妞ゅ湱娲版ê閿涙碍澧界悰灞炬纯閺傜増鎼锋担婊愮礄閸掓稑缂撻弬澧楅張顒婄礆



                log.warn("妞ゅ湱娲伴崥宥囆瀃{}]瀹告彃鐡ㄩ崷顭掔礉鐏忓棙澧界悰灞炬纯閺傜増鎼锋担?, excelProjectName);



                Ontology current = getCurrentVersion(excelProjectName);



                if (current == null) {



                    throw new RuntimeException("妞ゅ湱娲拌ぐ鎾抽悧鍫熸拱娑撳秴鐡ㄩ崷顭掔礉閺冪姵纭堕弴瀛樻煀");



                }







                // 娴燛xcel鐟欙絾鐎介弴瀛樻煀娣団剝



                Ontology updateEntity = new Ontology();



                updateEntity.setId(current.getId());



                if (firstDataRow != null && firstDataRow.getCell(3) != null) {



                    updateEntity.setNamespaceUri(firstDataRow.getCell(3).getStringCellValue());



                }



                return updateById(updateEntity);  // 鐠嬪啰鏁ら弴瀛樻煀閺傜《閸掓稑缂撻弬澧楅張?



            } else {



                // 妞ゅ湱娲版稉宥呯摠閸︻煉绱伴幍褑顢戦弬鏉款杻閹垮秳



                log.info("妞ゅ湱娲伴崥宥囆瀃{}]娑撳秴鐡ㄩ崷顭掔礉鐏忓棙澧界悰灞炬煀婢х偞鎼锋担?, excelProjectName);



                // 鐟欙絾鐎介崨钘夋倳缁屾椽妫块敍鍫ョ帛鐠併倕?妞ゅ湱娲伴崥宥忕礆



                String namespaceUri = "http://example.com/ontology/" + excelProjectName;



                if (firstDataRow != null && firstDataRow.getCell(3) != null) {



                    namespaceUri = firstDataRow.getCell(3).getStringCellValue();



                }







                // 閸掓稑缂撻弬鐗堟拱?



                Ontology ontology = new Ontology();



                ontology.setProjectName(excelProjectName);



                ontology.setCreator(creator);



                ontology.setNamespaceUri(namespaceUri);



                ontology.setVersionNumber("1.0");  // 閸掓繂顫愰悧鍫熸拱

    private String generateNewVersion(String currentVersion) {



        try {



            String[] parts = currentVersion.split("\\.");



            if (parts.length == 2) {



                int major = Integer.parseInt(parts[0]);



                int minor = Integer.parseInt(parts[1]);



                minor++;



                if (minor >= 10) {  //  minor娴ｅ秵10?



                    minor = 0;



                    major++;



                }



                return major + "." + minor;



            }



        } catch (Exception e) {



            log.error("閻楀牊婀伴崣袙閺嬫劕銇戠拹绱濇担璺ㄦ暏姒涱吇鎺戭杻", e);



        }



        return currentVersion + ".1";  // 鐟欙絾鐎芥径瑙﹂弮鍓佹畱闂勫秶楠囨径



    }

                    .eq(Ontology::getVersionStatus, 1)

        Map<String, Object> jsonData = new HashMap<>();



        jsonData.put("ontologyId", ontology.getId());



        jsonData.put("projectName", ontology.getProjectName());



        jsonData.put("version", ontology.getVersionNumber());



        jsonData.put("namespaceUri", ontology.getNamespaceUri());



        jsonData.put("exportTime", new SimpleDateFormat("yyyy-MM-dd HH:mm:ss").format(new Date()));



        



        // 閼惧嘲褰囬崶鎹愭皑閺佺増



        if (graphRepository.namedGraphExists(namedGraphUri)) {



            OntologyVisualizationDTO vizData = graphRepository.getVisualizationData(namedGraphUri);



            jsonData.put("nodes", vizData.getNodes());



            jsonData.put("edges", vizData.getEdges());



            jsonData.put("statistics", vizData.getStatistics());



        } else {



            jsonData.put("nodes", new ArrayList<>());



            jsonData.put("edges", new ArrayList<>());



        }



        



        // 閸愭瑥鍙嗛崫宥呯安



        try (OutputStream os = response.getOutputStream();



             PrintWriter writer = new PrintWriter(new OutputStreamWriter(os, StandardCharsets.UTF_8))) {



            ObjectMapper mapper = new ObjectMapper();



            writer.write(mapper.writerWithDefaultPrettyPrinter().writeValueAsString(jsonData));



            writer.flush();



            log.info("JSON鐎电厧閹存劕");



        }



    }

        String filename = ontology.getProjectName() + "_" + ontology.getVersionNumber() + "." + format.getExtension();



        response.setContentType(format.getMimeType());



        response.setHeader("Content-Disposition", "attachment;filename=" + 



                URLEncoder.encode(filename, StandardCharsets.UTF_8.name()));







        // 閸掓稑缂撶粻閸楁洜娈慠DF鈥



        Model model = ModelFactory.createDefaultModel();



        String ns = ontology.getNamespaceUri() != null ? ontology.getNamespaceUri() : 



                "http://example.org/ontology/";



        



        Resource ontologyResource = model.createResource(ns + ontology.getProjectName())



                .addProperty(RDF.type, model.createResource("http://www.w3.org/2002/07/owl#Ontology"))



                .addProperty(RDFS.label, ontology.getProjectName())



                .addProperty(model.createProperty(ns + "version"), ontology.getVersionNumber())



                .addProperty(model.createProperty(ns + "creator"), ontology.getCreator());







        try (OutputStream os = response.getOutputStream()) {



            model.write(os, format.getJenaLang().getName());



        }



    }

    @Override



    public List<ClassInfo> getOntologyClasses(Long id, String keyword) {



        Ontology ontology = baseMapper.selectById(id);



        if (ontology == null) {



            return new ArrayList<>();



        }







        String namedGraphUri = graphRepository.buildNamedGraphUri(



                ontology.getProjectName(), ontology.getVersionNumber());







        List<ClassInfo> classes = graphRepository.getClasses(namedGraphUri);



        



        // 閺嶅祦閸忔娊鐠囧秷绻?



        if (keyword != null && !keyword.trim().isEmpty()) {



            String lowerKeyword = keyword.toLowerCase();



            classes = classes.stream()



                    .filter(cls -> {



                        String name = cls.getName() != null ? cls.getName().toLowerCase() : "";



                        String description = cls.getDescription() != null ? cls.getDescription().toLowerCase() : "";



                        return name.contains(lowerKeyword) || description.contains(lowerKeyword);



                    })



                    .collect(Collectors.toList());



        }



        



        // 鐠佸墽鐤嗛悥鍓侀崥宥囆為敍鍫㈢暆閸栨牕顦╅悶鍡绱濈偤鎼存棁顕氭禒搴℃禈閺佺増宓佹惔鎾寸叀鐠囶澁



        for (ClassInfo cls : classes) {



            if (cls.getParentId() == null || cls.getParentId().isEmpty()) {



                // 鐏忔繆閺屻儲澹橀悥鍓侀敍rdfs:subClassOf?



                cls.setParentName("Thing");



            }



        }



        



        return classes;



    }

    /**



     * 娴燤ySQL閺屻儴顕楅崣顖濐潒閸栨牗鏆熼幑?



     */



    private OntologyVisualizationDTO getVisualizationDataFromMySQL(Integer ontologyId) {



        List<OntologyVisualizationDTO.NodeDTO> nodes = new ArrayList<>();



        List<OntologyVisualizationDTO.EdgeDTO> edges = new ArrayList<>();



        



        // 閺屻儴顕楃猾鏄忓Ν?



        List<OntologyClass> classes = ontologyClassMapper.selectByOntologyId(ontologyId);



        Map<Integer, String> classIdMap = new HashMap<>();



        for (OntologyClass cls : classes) {



            String nodeId = "class_" + cls.getClassId();



            classIdMap.put(cls.getClassId(), nodeId);



            nodes.add(OntologyVisualizationDTO.NodeDTO.builder()



                    .id(nodeId)



                    .label(cls.getName())



                    .type("class")



                    .color("#1890ff")



                    .size(60)



                    .build());



        }



        



        // 閺屻儴顕楃偘绶ラ懞鍌滃仯



        List<OntologyIndividual> individuals = ontologyIndividualMapper.selectByOntologyId(ontologyId);



        Map<Integer, String> individualIdMap = new HashMap<>();



        for (OntologyIndividual ind : individuals) {



            String nodeId = "ind_" + ind.getIndividualId();



            individualIdMap.put(ind.getIndividualId(), nodeId);



            nodes.add(OntologyVisualizationDTO.NodeDTO.builder()



                    .id(nodeId)



                    .label(ind.getName())



                    .type("individual")



                    .color("#52c41a")



                    .size(40)



                    .build());



        }



        



        // 閺屻儴顕楃猾鑽ゆ埛閹靛灝鍙х化?



        for (OntologyClass cls : classes) {



            if (cls.getParentClassId() != null && classIdMap.containsKey(cls.getParentClassId())) {



                edges.add(OntologyVisualizationDTO.EdgeDTO.builder()



                        .id("edge_sub_" + cls.getClassId())



                        .source("class_" + cls.getClassId())



                        .target("class_" + cls.getParentClassId())



                        .label("subClassOf")



                        .type("subClassOf")



                        .color("#999999")



                        .build());



            }



        }



        



        // 閺屻儴顕楃偘-璇插彠?



        for (OntologyIndividual ind : individuals) {



            List<IndividualType> types = individualTypeMapper.selectByIndividualId(ind.getIndividualId());



            for (IndividualType type : types) {



                if (classIdMap.containsKey(type.getClassId())) {



                    edges.add(OntologyVisualizationDTO.EdgeDTO.builder()



                            .id("edge_type_" + ind.getIndividualId())



                            .source("ind_" + ind.getIndividualId())



                            .target("class_" + type.getClassId())



                            .label("instanceOf")



                            .type("instanceOf")



                            .color("#52c41a")



                            .build());



                }



            }



        }



        



        // 閺屻儴顕楃偘绶ラ梻鏉戝彠?



        for (OntologyIndividual ind : individuals) {



            List<IndividualPropertyValue> properties = 



                individualPropertyValueMapper.selectByIndividualId(ind.getIndividualId());



            for (IndividualPropertyValue prop : properties) {



                if (prop.getObjectIndividualId() != null && 



                    individualIdMap.containsKey(prop.getObjectIndividualId())) {



                    edges.add(OntologyVisualizationDTO.EdgeDTO.builder()



                            .id("edge_rel_" + prop.getId())



                            .source("ind_" + ind.getIndividualId())



                            .target("ind_" + prop.getObjectIndividualId())



                            .label("relatedTo")



                            .type("objectProperty")



                            .color("#fa8c16")



                            .build());



                }



            }



        }



        



        log.info("getVisualizationDataFromMySQL: found {} nodes, {} edges", nodes.size(), edges.size());



        



        return OntologyVisualizationDTO.builder()



                .nodes(nodes)



                .edges(edges)



                .build();



    }



    



    @Override



    @Transactional(rollbackFor = Exception.class)



    public GraphWarehouseResponseDTO warehouseGraph(GraphWarehouseDTO dto) {



        log.info("warehouseGraph: 娴樼拫鍗炲弳鎼存挸鍩孧ySQL閿沷ntologyId={}, ontologyName={}", 



                dto.getOntologyId(), dto.getOntologyName());



        



        try {



            // 1. 閼惧嘲褰囬張顑跨秼娣団剝



            Ontology ontology = baseMapper.selectById(dto.getOntologyId());



            if (ontology == null) {



                throw new RuntimeException("閺堫兛缍嬫稉宥呯摠閸︻煉绱漣d=" + dto.getOntologyId());



            }



            



            // 2. 閸掓稑缂撻弬澧楅張顒傛畱閺堫兛缍嬬拋鏉跨秿



            Ontology newVersion = new Ontology();



            BeanUtils.copyProperties(ontology, newVersion);



            newVersion.setId(null);



            newVersion.setVersionNumber(dto.getVersion());



            newVersion.setVersionStatus(1); // 瑜版挸澧犻悧鍫熸拱



            newVersion.setCreateTime(new Date());



            newVersion.setModifyTime(new Date());



            



            // 娣囨繂鐡ㄩ弬澧楅張?



            baseMapper.insert(newVersion);



            Integer newOntologyId = newVersion.getId();



            



            // 3. 鐏忓棙閻楀牊婀伴弽顔囨稉鍝勫坊閸欒尙澧楅張?



            ontology.setVersionStatus(0);



            baseMapper.updateById(ontology);



            



            // 4. 娣囨繂鐡ㄩ懞鍌滃仯閸扢ySQL



            Map<String, Integer> nodeIdMap = new HashMap<>(); // 閸撳秶顏琁D -> 閺佺増宓佹惔ID



            



            // 娣囨繂鐡ㄧ猾鏄忓Ν?



            for (GraphWarehouseDTO.NodeDTO node : dto.getNodes()) {



                if ("class".equals(node.getType())) {



                    OntologyClass ontologyClass = new OntologyClass();



                    ontologyClass.setOntologyId(newOntologyId);



                    ontologyClass.setName(node.getName());



                    ontologyClass.setDescription(node.getDescription());



                    ontologyClass.setCreateTime(new Date());



                    ontologyClass.setUpdateTime(new Date());



                    



                    ontologyClassMapper.insert(ontologyClass);



                    nodeIdMap.put(node.getId(), ontologyClass.getClassId());



                }



            }



            



            // 娣囨繂鐡ㄧ偘绶ラ懞鍌滃仯



            for (GraphWarehouseDTO.NodeDTO node : dto.getNodes()) {



                if ("individual".equals(node.getType())) {



                    OntologyIndividual individual = new OntologyIndividual();



                    individual.setOntologyId(newOntologyId);



                    individual.setName(node.getName());



                    individual.setDescription(node.getDescription());



                    individual.setCreateTime(new Date());



                    individual.setUpdateTime(new Date());



                    



                    ontologyIndividualMapper.insert(individual);



                    nodeIdMap.put(node.getId(), individual.getIndividualId());



                }



            }



            



            // 5. 娣囨繂鐡ㄩ崗宕囬兇閸扢ySQL



            int relationCount = 0;



            for (GraphWarehouseDTO.EdgeDTO edge : dto.getRelationships()) {



                Integer sourceId = nodeIdMap.get(edge.getSource());



                Integer targetId = nodeIdMap.get(edge.getTarget());



                



                if (sourceId == null || targetId == null) {



                    log.warn("閸忓磭閮撮懞鍌滃仯閺堫亝澹橀崚? {} -> {}", edge.getSource(), edge.getTarget());



                    continue;



                }



                



                // 閸掋倖鏌囬崗宕囬兇璇



                String sourceType = getNodeType(dto.getNodes(), edge.getSource());



                String targetType = getNodeType(dto.getNodes(), edge.getTarget());



                



                if ("class".equals(sourceType) && "class".equals(targetType)) {



                    // 鑽ゆ埛閹靛灝鍙х化?



                    ClassInheritance inheritance = new ClassInheritance();



                    inheritance.setChildClassId(sourceId);



                    inheritance.setParentClassId(targetId);



                    classInheritanceMapper.insert(inheritance);



                    relationCount++;



                } else if ("individual".equals(sourceType) && "class".equals(targetType)) {



                    // 鐎圭偘绶ョ猾璇查崗宕囬兇



                    IndividualType individualType = new IndividualType();



                    individualType.setIndividualId(sourceId);



                    individualType.setClassId(targetId);



                    individualTypeMapper.insert(individualType);



                    relationCount++;



                } else if ("individual".equals(sourceType) && "individual".equals(targetType)) {



                    // 鐎圭偘绶ヨ杽鐏炵偞褍鍙х化?



                    IndividualPropertyValue propertyValue = new IndividualPropertyValue();



                    propertyValue.setIndividualId(sourceId);



                    propertyValue.setPropertyId(1); // 姒涱吇鐏炵偞顪廌



                    propertyValue.setValueType("OBJECT");



                    propertyValue.setObjectIndividualId(targetId);



                    individualPropertyValueMapper.insert(propertyValue);



                    relationCount++;



                }



            }



            



            log.info("warehouseGraph: 閸忋儱绨遍幋濮涢敍灞肩箽鐎涙ü {} 娑擃亣閻愮櫢绱漿} 閺夊彠璇插煂MySQL", 



                    nodeIdMap.size(), relationCount);



            



            return GraphWarehouseResponseDTO.builder()



                    .status("success")



                    .message("閸ユ崘姘ㄩ崗銉ョ氨閹存劕")



                    .nodeCount(nodeIdMap.size())



                    .relationshipCount(relationCount)



                    .version(dto.getVersion())



                    .timestamp(LocalDateTime.now())



                    .build();



                    



        } catch (Exception e) {



            log.error("warehouseGraph: 閸忋儱绨辨径", e);



            throw new RuntimeException("閸ユ崘姘ㄩ崗銉ョ氨婢惰精: " + e.getMessage(), e);



        }



    }



    



    /**



     * 閼惧嘲褰囬懞鍌滃仯璇



     */



    private String getNodeType(List<GraphWarehouseDTO.NodeDTO> nodes, String nodeId) {



        for (GraphWarehouseDTO.NodeDTO node : nodes) {



            if (node.getId().equals(nodeId)) {



                return node.getType();



            }



        }



        return "individual";



    }
